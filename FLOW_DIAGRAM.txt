                        🔄 AUTO-DRAW GAME FLOW
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

    Game Created
      │
      ▼
    [START GAME] Button Clicked
      │ ├─ gameState.autoDrawEnabled = true
      │ ├─ startAutoDrawTimer() called
      │ └─ Button changes to [PAUSE]
      │
      ├─────────────────┐
      │                 │
      ▼                 │
    DRAW FIRST NUMBER   │
      ├─ Random number 1-90 (unused)  │
      ├─ drawAuto() function          │
      ├─ Send via WebSocket           │
      ├─ Voice announce number        │
      │                               │
      ├─ WAIT 6.5 SECONDS ◄───────────┘
      │
      ▼
    DRAW NEXT NUMBER
      ├─ Random number 1-90 (unused)
      ├─ Send via WebSocket
      ├─ Auto-mark on all player tickets
      ├─ Voice announce
      │
      ├─ [PAUSE] Button Available
      │   └─ Stops drawing, shows [RESUME]
      │
      ├─ [RESUME] Button Available
      │   └─ Resumes drawing, shows [PAUSE]
      │
      └─ Continue until...
         ├─ Admin clicks PAUSE, or
         ├─ All 90 numbers called, or
         └─ Game ends

                        🏆 WIN VERIFICATION & TIE DETECTION
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

    Player Claims Win
      │ ├─ Claim Type: First Line / Two Lines / Full House
      │ ├─ Send claim_win message
      │ └─ Mark claim in database
      │
      ▼
    Server Validates Win
      ├─ Check if all required numbers are marked
      ├─ Verify win type conditions
      │
      ├─ WIN VALID
      │  │
      │  ├─ Add to winners list
      │  ├─ Check for OTHER winners with SAME win type
      │  │
      │  ├─ TIE DETECTED (2+ players same prize)
      │  │  │
      │  │  ├─ Announce: "TIE x2" (2 winners)
      │  │  ├─ Show all tied player names
      │  │  ├─ Auto-calculate: Prize / Number of Winners
      │  │  ├─ Display to all players and admin
      │  │  └─ No manual intervention needed
      │  │
      │  └─ NO TIE (Only this player won)
      │     ├─ Announce: "[Player Name] wins [Prize Type]!"
      │     └─ Show win details
      │
      └─ WIN INVALID
         ├─ Show error: "Invalid win claim"
         ├─ Numbers don't match requirements
         └─ Allow retry
